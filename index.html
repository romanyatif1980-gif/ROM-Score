// ====== Firebase Configuration ====== const firebaseConfig = { apiKey: "YOUR_API_KEY", authDomain: "YOUR_PROJECT_ID.firebaseapp.com", databaseURL: "https://YOUR_PROJECT_ID-default-rtdb.firebaseio.com", projectId: "YOUR_PROJECT_ID", storageBucket: "YOUR_PROJECT_ID.appspot.com", messagingSenderId: "YOUR_SENDER_ID", appId: "YOUR_APP_ID" }; firebase.initializeApp(firebaseConfig); const db = firebase.database(); // ====== Toast Function ====== function showToast(message) { const toast = document.getElementById('toast'); toast.innerText = message; toast.style.opacity = 1; setTimeout(() => toast.style.opacity = 0, 3000); } // ====== Utility: Generate ID ====== function generateID() { return 'id-' + Date.now() + '-' + Math.floor(Math.random() * 1000); } // ====== Create Game ====== async function createGame() { const gameID = generateID(); const gameData = { type: "Domino", status: "ongoing", createdAt: Date.now(), players: {}, rounds: {} }; await db.ref('games/' + gameID).set(gameData); showToast('تم إنشاء جيم جديد'); } // ====== Fetch and Display Games ====== const gamesContainer = document.getElementById('gamesContainer'); function displayGames(snapshot) { gamesContainer.innerHTML = ''; const games = snapshot.val(); if (!games) return; for (const gameID in games) { const game = games[gameID]; const card = document.createElement('div'); card.className = 'game-card'; let playersHTML = ''; for (const pid in game.players) { playersHTML += ``; } playersHTML += '
اللاعبالسكور
${game.players[pid].name}${game.players[pid].score || 0}
'; card.innerHTML = `

جيم: ${game.type} (${game.status})

${playersHTML}
`; gamesContainer.appendChild(card); } } // ====== Listen to Firebase Changes ====== db.ref('games').on('value', displayGames); // ====== Add Player ====== async function addPlayer(gameID) { const name = prompt('أدخل اسم اللاعب:'); if (!name) return; const playerID = generateID(); await db.ref(`games/${gameID}/players/${playerID}`).set({ name, score: 0 }); showToast('تم إضافة اللاعب'); } // ====== Add Round ====== async function addRound(gameID) { const roundID = 'round-' + Date.now(); const gameRef = db.ref(`games/${gameID}/players`); const snapshot = await gameRef.get(); const players = snapshot.val(); if (!players) return; const scores = {}; for (const pid in players) { const score = parseInt(prompt(`اللاعب ${players[pid].name} : السكور`)) || 0; scores[pid] = score; await db.ref(`games/${gameID}/players/${pid}/score`).set((players[pid].score || 0) + scores[pid]); } await db.ref(`games/${gameID}/rounds/${roundID}`).set({ scores, createdAt: Date.now() }); showToast('تمت إضافة الجولة'); } // ====== Delete Game ====== async function deleteGame(gameID) { const pass = prompt('أدخل الرقم السري للحذف:'); if (pass !== '1234') return alert('رقم سري خاطئ'); await db.ref('games/' + gameID).remove(); showToast('تم حذف الجيم'); } // ====== Download Excel ====== async function downloadExcel() { const snapshot = await db.ref('games').get(); const games = snapshot.val(); if (!games) return alert('لا توجد بيانات'); const wb = XLSX.utils.book_new(); for (const gid in games) { const game = games[gid]; const wsData = [['اللاعب', 'السكور']]; for (const pid in game.players) { wsData.push([game.players[pid].name, game.players[pid].score]); } const ws = XLSX.utils.aoa_to_sheet(wsData); XLSX.utils.book_append_sheet(wb, ws, game.type); } XLSX.writeFile(wb, 'Tana_Rona_Scores.xlsx'); showToast('تم تحميل Excel'); }
